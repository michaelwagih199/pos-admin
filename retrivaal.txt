@Component({
  selector: 'app-retrivals',
  templateUrl: './retrivals.component.html',
  styleUrls: ['./retrivals.component.scss'],
})
export class RetrivalsComponent implements OnInit {
  displayedColumns: string[] = [
    'id',
    'retrievalsCode',
    'totalCost',
    'discount',
    'netCost',
    'person',          // ← changed
    'createdDate',
    'actions',
  ];
  arabic: Arabic = new Arabic();
  retrivalsList!: RetrivalsModel[];
  retriveModel: RetrivalsModel = new RetrivalsModel();
  isLoading = false;
  constructor(
    private _snackBar: MatSnackBar,
    private retrivalService: RetrivalsService,
    private retrivalDetailsService: RetrivalsDetailsService,
    private router: Router,
    private dialog: MatDialog) { }
  ngOnInit(): void {
    this.retrieve();
  }
  /**
   * data
   */
  retrieve() {
    this.isLoading = true;
    const params = this.getRequestParams(this.page, this.pageSize);
    this.retrivalService.getAllPagination(params).subscribe(
      (data) => {
        this.isLoading = false;
        this.retrivalsList = data.retrievalsBills;
        this.count = data.totalItems;
      },
      (error) => {
        this.isLoading = false;
        console.log(error);
      }
    );
  }
  /**
   * evants
   */
  editeDialog(obj: any) { }
  deleteDialog(obj: RetrivalsModel) {
    const dialogRef = this.dialog.open(ConfirmationDialog, {
      data: {
        message: `${this.arabic.stock.category.util.dialog.deleteDialog.title}: ${obj.retrievalsCode}`,
        buttonText: {
          ok: `${this.arabic.stock.category.util.dialog.dialogButtons.ok}`,
          cancel: `${this.arabic.stock.category.util.dialog.dialogButtons.cancel}`
        }
      }
    });
    dialogRef.afterClosed().subscribe((confirmed: boolean) => {
      if (confirmed) {
        this.retrivalService.delete(obj.id).subscribe(data => {
          this.openSnackBar(`${this.arabic.stock.category.util.dialog.notification.deleted}`, '')
          this.retrieve()
        }, error => console.log(error))
        const a = document.createElement('a');
        a.click();
        a.remove();
      }
    });
  }
  Ondetails(obj: any) {
    this.dialog.open(RerivalDetailsComponent, {
      data: { id: obj.id },
    });
  }
  /**
   * ui ux
   */
  openSnackBar(message: string, action: string) {
    this._snackBar.open(message, action, {
      duration: 2000,
    });
  }
  page = 1;
  count = 0;
  pageSize = 6;
  handlePageChange(event: any) {
    this.page = event;
    this.retrieve();
  }
  getRequestParams(page: any, pageSize: any) {
    // tslint:disable-next-line:prefer-const
    let params: any = {};
    if (page) {
      params[`page`] = page - 1;
    }
    if (pageSize) {
      params[`size`] = pageSize;
    }
    return params;
  }
  addDialog() {
    this.router.navigate(['/retrivals/new']);
  }
}

import { Component, Inject, OnInit } from '@angular/core';
import {
  AbstractControl, FormBuilder, FormGroup,
  ValidationErrors, ValidatorFn, Validators
} from '@angular/forms';
import { MatDialogRef, MAT_DIALOG_DATA } from '@angular/material/dialog';
import { RetrivalsDetailsModel } from '../../models/retrivals-details-model';
import { ReturnableProduct } from '../../models/returnable-product';
import { ReturnType } from '../../models/retrivals-model';
interface DialogData {
  product: ReturnableProduct;
  returnType: ReturnType;
}
@Component({
  selector: 'app-return-product-dialog',
  templateUrl: './return-product-dialog.component.html',
  styleUrls: ['./return-product-dialog.component.scss'],
})
export class ReturnProductDialogComponent implements OnInit {
  form!: FormGroup;
  reasons = [
    { value: 'DEFECT', label: 'عيب صناعة' },
    { value: 'WRONG_ITEM', label: 'منتج خاطئ' },
    { value: 'EXPIRED', label: 'منتهي الصلاحية' },
    { value: 'NOT_NEEDED', label: 'غير مطلوب' },
    { value: 'OTHER', label: 'أخرى' },
  ];
  maxQty = 0;
  maxUnits = 0;
  multiInvoiceInvoices: Array<any> = [];
  constructor(
    private fb: FormBuilder,
    private ref: MatDialogRef<ReturnProductDialogComponent>,
    @Inject(MAT_DIALOG_DATA) public data: DialogData
  ) { }
  ngOnInit(): void {
    const p = this.data.product;
    this.multiInvoiceInvoices = p.invoices ?? [];
    const firstAvailable = this.multiInvoiceInvoices.find(
      (inv) => Number(inv.returnableQty) > 0
    );
    const initialInvoice = firstAvailable ?? this.multiInvoiceInvoices[0];
    // Rolls ceiling
    this.maxQty = Number(initialInvoice?.returnableQty ?? p.availableCanvases ?? 0);
    // Units ceiling — comes from the DTO, NOT computed from rolls
    this.maxUnits = Number(
      (initialInvoice as any)?.returnableUnits
      ?? (p as any).availableUnits
      ?? 0
    );
    // Default per-unit value = the whole remaining units (user can lower it)
    const defaultUnitValue = this.maxUnits > 0 ? this.maxUnits : 1;
    const rollsMin = this.maxQty > 0 ? 1 : 0;
    const rollsDefault = this.maxQty > 0 ? 1 : 0;
    this.form = this.fb.group(
      {
        sourceDetailId: [
          initialInvoice?.saleOrderDetailId
          ?? initialInvoice?.purchasesBillDetailId
          ?? null,
        ],
        quantity: [
          rollsDefault,
          [Validators.required, Validators.min(rollsMin), Validators.max(this.maxQty)],
        ],
        unitNumber: [
          defaultUnitValue,
          [Validators.required, Validators.min(0.001), Validators.max(this.maxUnits)],
        ],
        condition: ['GOOD', Validators.required],
        reason: ['', Validators.required],
        notes: [''],
      },
      { validators: this.returnableValidator() }
    );
    this.form.get('sourceDetailId')?.valueChanges.subscribe((id) => {
      const inv = this.multiInvoiceInvoices.find(
        (i) => (i.saleOrderDetailId ?? i.purchasesBillDetailId) === id
      );
      if (!inv) return;
      this.maxQty = Number(inv.returnableQty) || 0;
      this.maxUnits = Number(inv.returnableUnits) || 0;
      const rollsMin = this.maxQty > 0 ? 1 : 0;      // ← add
      const qtyCtrl = this.form.get('quantity');
      const unitCtrl = this.form.get('unitNumber');
      qtyCtrl?.setValidators([
        Validators.required,
        Validators.min(rollsMin),                    // ← was min(1)
        Validators.max(this.maxQty),
      ]);
      unitCtrl?.setValidators([
        Validators.required,
        Validators.min(0.001),
        Validators.max(this.maxUnits),
      ]);
      qtyCtrl?.updateValueAndValidity();
      unitCtrl?.updateValueAndValidity();
      this.form.updateValueAndValidity();
    });
  }
  private returnableValidator(): ValidatorFn {
    return (group: AbstractControl): ValidationErrors | null => {
      const qty = Number(group.get('quantity')?.value ?? 0);
      const unit = Number(group.get('unitNumber')?.value ?? 0);
      const errs: ValidationErrors = {};
      // Rolls and units are INDEPENDENT ceilings — no multiplication
      if (qty > this.maxQty) errs['quantityOverflow'] = true;
      if (unit > this.maxUnits) errs['unitsOverflow'] = true;
      return Object.keys(errs).length ? errs : null;
    };
  }
  get totalUnits(): number {
    return Number(this.form?.get('unitNumber')?.value ?? 0);
  }
  get remainingUnits(): number {
    return Math.max(0, this.maxUnits - this.totalUnits);
  }
  get selectedInvoiceCode(): string | null {
    const id = this.form?.get('sourceDetailId')?.value;
    const inv = this.multiInvoiceInvoices.find(
      (i) => (i.saleOrderDetailId ?? i.purchasesBillDetailId) === id
    );
    return inv?.orderCode ?? null;
  }
  get measurementUnitLabel(): string {
    return this.data.product.measurementUnit === 'METER' ? 'متر' : 'كجم';
  }
  save(): void {
    if (this.form.invalid) {
      this.form.markAllAsTouched();
      return;
    }
    const v = this.form.value;
    const p = this.data.product;
    const unitPrice = p.lastPrice ?? 0;
    const line: RetrivalsDetailsModel = {
      id: 0,
      productId: p.productId,
      productName: p.productName,
      productCode: p.productCode ?? '',
      quantity: v.quantity,
      unitNumber: v.unitNumber,
      price: unitPrice,
      total: v.unitNumber * unitPrice,
      returnCondition: v.condition,
      returnReason: v.reason,
      sourceSaleOrderDetailId:
        this.data.returnType === 'CUSTOMER'
          ? v.sourceDetailId ?? undefined
          : undefined,
      sourcePurchasesBillDetailId:
        this.data.returnType === 'SUPPLIER'
          ? v.sourceDetailId ?? undefined
          : undefined,
    };
    this.ref.close(line);
  }
  close(): void {
    this.ref.close();
  }
}

import { Component, OnInit } from '@angular/core';
import { FormControl } from '@angular/forms';
import { MatDialog } from '@angular/material/dialog';
import { MatSnackBar } from '@angular/material/snack-bar';
import { Router } from '@angular/router';                                  // 👈 add
import { Observable, of } from 'rxjs';
import { debounceTime, distinctUntilChanged, switchMap, map, catchError, startWith } from 'rxjs/operators';
import { CustomerService } from 'src/app/customers/service/customer.service';
import { SupliersService } from 'src/app/suppliers/service/supliers.service';
import { RetrivalsDetailsModel } from '../../models/retrivals-details-model';
import { RetrivalsModel, ReturnType, RefundMethod } from '../../models/retrivals-model';
import { ReturnableProduct } from '../../models/returnable-product';
import { RetrivalsService } from '../../service/retrivals.service';
import { ReturnProductDialogComponent } from '../return-product-dialog/return-product-dialog.component';
interface Person { id: number; name: string; phone?: string; }
@Component({
  selector: 'app-retrieval-wizard',
  templateUrl: './retrieval-wizard.component.html',
  styleUrls: ['./retrieval-wizard.component.scss'],
})
export class RetrievalWizardComponent implements OnInit {
  step = 0;
  returnType: ReturnType = 'CUSTOMER';
  selectedPerson: Person | null = null;
  searchCtrl = new FormControl();
  filteredPeople: Observable<Person[]> = of([]);
  products: ReturnableProduct[] = [];
  loadingProducts = false;
  cart: RetrivalsDetailsModel[] = [];
  discount = 0;
  refundMethod: RefundMethod = 'CREDIT';
  saving = false;
  reviewColumns: string[] = ['product', 'pieces', 'units', 'condition', 'total'];
  constructor(
    private dialog: MatDialog,
    private router: Router,                    // 👈 was MatDialogRef
    private snack: MatSnackBar,
    private customerService: CustomerService,
    private supplierService: SupliersService,
    private retrievalService: RetrivalsService
  ) { }
  ngOnInit(): void {
    this.filteredPeople = this.searchCtrl.valueChanges.pipe(
      startWith(''),
      debounceTime(250),
      distinctUntilChanged(),
      switchMap((term: string) => {
        const q = (term ?? '').trim();
        if (q.length < 2) return of([]);
        const obs$ = this.returnType === 'CUSTOMER'
          ? this.customerService.findByName(q)
          : this.supplierService.findByName(q);
        return obs$.pipe(catchError(() => of([])));
      }),
      map((res: any) => this.normalizePeople(res))
    );
  }
  private normalizePeople(res: any): Person[] {
    const list: any[] = Array.isArray(res) ? res : (res?.data ?? []);
    return list.map((p) => ({
      id: p.id,
      name: p.customerName ?? p.supplierName ?? '',
      phone: p.customerPhone ?? p.supplierPhone ?? '',
    }));
  }
  onPersonSelected(p: Person) {
    this.selectedPerson = p;
    this.step = 2;
    this.loadReturnable();
  }
  loadReturnable() {
    if (!this.selectedPerson) return;
    this.loadingProducts = true;
    this.retrievalService.getReturnableProducts(this.selectedPerson.id, this.returnType).subscribe({
      next: (res) => {
        const list: ReturnableProduct[] = res?.data ?? [];
        this.products = list.filter(p =>
          (p.availableCanvases ?? 0) > 0 || (p.availableUnits ?? 0) > 0
        );
        this.loadingProducts = false;
        if (this.products.length === 0) {
          this.snack.open('لا توجد منتجات قابلة للإرجاع لهذا الشخص', '', { duration: 3000 });
        }
      },
      error: () => (this.loadingProducts = false),
    });
  }
  openReturnDialog(p: ReturnableProduct) {
    const ref = this.dialog.open(ReturnProductDialogComponent, {
      width: '560px',
      disableClose: true,
      data: { product: p, returnType: this.returnType },
    });
    ref.afterClosed().subscribe((line?: RetrivalsDetailsModel) => {
      if (!line) return;
      const i = this.cart.findIndex(l =>
        l.productId === line.productId
        && l.returnCondition === line.returnCondition
        && l.sourceSaleOrderDetailId === line.sourceSaleOrderDetailId
        && l.sourcePurchasesBillDetailId === line.sourcePurchasesBillDetailId
      );
      if (i > -1) {
        this.cart[i].quantity += line.quantity;      // rolls — metadata
        this.cart[i].unitNumber += line.unitNumber;    // units — accumulate
        this.cart[i].total = this.cart[i].unitNumber * this.cart[i].price;
      } else {
        this.cart.push(line);
      }
    });
  }
  removeLine(i: number) { this.cart.splice(i, 1); }
  get subtotal() { return this.cart.reduce((s, l) => s + (l.total || 0), 0); }
  get netTotal() { return Math.max(0, this.subtotal - (this.discount || 0)); }
  submit() {
    if (!this.selectedPerson || this.cart.length === 0) return;
    this.saving = true;
    const model: RetrivalsModel = {
      id: 0,
      retrievalsCode: '',
      totalCost: this.subtotal,
      discount: this.discount || 0,
      netCost: this.netTotal,
      createdDate: '',
      returnType: this.returnType,
      refundMethod: this.refundMethod,
      customer: this.returnType === 'CUSTOMER' ? (this.selectedPerson as any) : null,
      supplier: this.returnType === 'SUPPLIER' ? (this.selectedPerson as any) : null,
    };
    const payload = {
      model,
      returnType: this.returnType,
      refundMethod: this.refundMethod,
      notes: '',
      details: this.cart.map(l => ({
        productId: l.productId,
        quantity: l.quantity,
        unitNumber: l.unitNumber,
        price: l.price,
        returnCondition: l.returnCondition,
        returnReason: l.returnReason,
        sourceSaleOrderDetailId: l.sourceSaleOrderDetailId,
        sourcePurchasesBillDetailId: l.sourcePurchasesBillDetailId,
      })),
    };
    this.retrievalService.createRetrieval(payload, this.selectedPerson.id, this.returnType).subscribe({
      next: (res) => {
        this.saving = false;
        const code = res?.data?.retrievalsCode ?? '';
        this.snack.open(`تم حفظ المرتجع: ${code}`, '', { duration: 3000 });
        this.router.navigate(['/retrivals']);            // 👈 was dialogRef.close(true)
      },
      error: (e) => {
        this.saving = false;
        this.snack.open(e?.error?.message || 'فشل الحفظ', '', { duration: 4000 });
      },
    });
  }
  close() { this.router.navigate(['/retrivals']); }       // 👈 was dialogRef.close(false)
}

### ./service/retrivals.service.ts
import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { Observable } from 'rxjs';
import { environment } from 'src/environments/environment';
import { ReturnType } from '../models/retrivals-model';
@Injectable({ providedIn: 'root' })
export class RetrivalsService {
  private baseUrl = `${environment.baseUrl}/retrievals`;
  constructor(private http: HttpClient) {}
  getAllPagination(params: any): Observable<any> {
    return this.http.get(`${this.baseUrl}/pageable`, { params });
  }
  getBillsCode(): Observable<any> {
    return this.http.get(`${this.baseUrl}/nextCode`);
  }
  getReturnableProducts(personId: number, type: ReturnType): Observable<any> {
    const params: any = type === 'CUSTOMER'
      ? { customerId: personId }
      : { supplierId: personId };
    return this.http.get(`${this.baseUrl}/returnable-products`, { params });
  }
  createRetrieval(payload: any, personId: number, type: ReturnType): Observable<any> {
    const params: any = type === 'CUSTOMER'
      ? { customerId: personId }
      : { supplierId: personId };
    return this.http.post(`${this.baseUrl}/createRetrivalBills`, payload, { params });
  }
  delete(id: number): Observable<any> {
    return this.http.put(`${this.baseUrl}/archive?id=${id}`, null);
  }
}
